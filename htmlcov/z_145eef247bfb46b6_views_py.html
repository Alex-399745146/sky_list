39def filter_aeroplanes(aeroplanes: Iterable[Aeroplane], reg_countries: list[str]) -> list[Aeroplane]: 

40 """Фильтрация самолётов по стране регистрации (origin_country)""" 

41 if not reg_countries: 

42 return list(aeroplanes) 

43 

44 reg_countries_norm = {c.strip().lower() for c in reg_countries if c.strip()}

48def parse_altitude_range(raw: str) -> tuple[float, float]: 

49 """Парсит строку диапазона высот вида '1000-2000' или '1000 - 2000'.""" 

50 raw = raw.strip() 

51 if not raw: 

52 # Пустая строка — без ограничений по высоте 

53 return float("-inf"), float("inf") 

54 

55 parts = raw.replace(" ", "").split("-") 

56 if len(parts) != 2: 

57 raise ValueError("Диапазон высот должен быть в формате: min-max") 

58 

59 low, high = map(float, parts) 

60 if low > high: 

61 low, high = high, low 

62 return low, high

65def get_aeroplanes_by_altitude(aeroplanes: Iterable[Aeroplane], altitude_range: str) -> list[Aeroplane]: 

66 """Фильтрация самолётов по диапазону высот.""" 

67 low, high = parse_altitude_range(altitude_range) 

68 return [plane for plane in aeroplanes if low <= plane.altitude <= high]

71def sort_aeroplanes(aeroplanes: Iterable[Aeroplane]) -> list[Aeroplane]: 

72 """ 

73 Сортировка самолётов, поскольку в Aeroplane реализованы __lt__/__eq__..., 

74 sorted() может использовать их напрямую.

76 return sorted(aeroplanes, reverse=True) # от большего к меньшему (высота, потом скорость)
